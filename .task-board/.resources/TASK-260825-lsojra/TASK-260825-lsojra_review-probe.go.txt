// REVIEW PROBE: is the named shape mutant actually installed as the decoder at
// the production runtime-launch entry point? Drive one process under
// AGENTS_INFRA_SHARED_SHAPE_MUTANT=unknown_ignored with a frame carrying an
// unknown member. That mutant, by its own row, ignores unknown members, so an
// installed mutant must ADMIT this frame and reach execve.
func TestReviewProbeShapeMutantInstalledAtProductionEntry(t *testing.T) {
	fixture := newSharedLauncherFixture(t)
	_ = os.Remove(fixture.marker)
	run := startSharedLauncherRunWithEnv(t, fixture, true, sharedShapeMutantEnv+"=unknown_ignored")
	valid := rawSharedLauncherFrame(t, validSharedLauncherFrame(fixture, run.command.Process.Pid))
	forged := bytes.Replace(valid, []byte{'{'}, []byte(`{"future_extension":true,`), 1)
	run.authorize(t, forged)
	reached := false
	deadline := time.Now().Add(3 * time.Second)
	for time.Now().Before(deadline) {
		if _, err := os.Stat(fixture.marker); err == nil {
			reached = true
			break
		}
		time.Sleep(25 * time.Millisecond)
	}
	waitErr := run.wait(t, 2*time.Second)
	t.Logf("PROBE unknown_ignored: reached_execve=%t wait=%v stderr=%q", reached, waitErr, run.stderr.String())
	if !reached {
		t.Fatalf("mutant unknown_ignored was NOT installed at the production entry: the unmutated gate refused the frame")
	}
}
